Версия: 2026-11-05. Основа: ZIP my-test-project-main.zip, две инструкции и шесть PHP-файлов.

ВНИМАНИЕ: Это инструкция для контролируемой миграции в НОВУЮ ПУСТУЮ ЦЕЛЕВУЮ БАЗУ, не для перезаписи действующей crm2. Скрипты 01/02/03/05 не являются полностью идемпотентными. После частичного импорта не перезапускать их без восстановления целевой базы. Остановить запись в старую CRM на время финального экспорта. Подготовить резервные копии и проверить восстановление. Без этого нельзя безопасно переключать прод.

Что изменено относительно архива: 04_import_clients.php теперь сразу заменяет legacy pos на текст по (dictionary_pkid, pos) и содержит четыре подтверждённых исключения. Поэтому массовый UPDATE через check_dictionary_mapping.php больше не требуется. Непонятные 13-16-значные значения сохраняются как есть и считаются в отчёте; при их наличии переключение на прод запрещено до расследования. Неизвестные соответствия НЕ угадываются. Прямое объединение по foreignPkid запрещено из-за конфликтов.

Проверенные результаты старого теста (не норма для нового прода): 201 база, 2050 полей, 109 справочников, 1449 вариантов, 740 разрешений, 973036 клиентов, 2981282 значения полей, 341896 контактов. В старом тесте 2073 некорректных строки клиентов, 1496 ссылок клиентов на отсутствующие базы, 7101 несопоставленных legacy значений списков. Эти проблемы не исчезнут автоматически из-за изменения скрипта.

Границы миграции: экспортируем database, client, list, databasePerm, clientHistory, relation_type. Импортируем первые четыре, контакты формируем из client.jsonl. История clientHistory и relation_type сохраняются только в JSONL; отдельного импортёра в архиве нет. actionHistory, crmInvoices, io_sequence не включены.

<?php
error_reporting(E_ALL);
ini_set("display_errors", "1");
set_time_limit(0);

require "/var/www/html/app/io/api.php";

$host = getenv("MONGO_HOST");
$user = getenv("MONGO_USER");
$pass = getenv("MONGO_PASSWORD");

if (!$host) {
    fwrite(STDERR, "ERROR: MONGO_HOST is empty\n");
    exit(1);
}

$connection = array(
    "server" => "mongodb://" . $host,
    "options" => array(
        "username" => $user,
        "password" => $pass
    )
);

ioSetMongoConnection("crm", $connection, $connection);

$err = null;
$mongo = ioGetMongoConnection("crm", "read", $err);

if (!$mongo) {
    fwrite(STDERR, "ERROR: MongoDB connection failed: " . $err . "\n");
    exit(1);
}

$db = $mongo->selectDB("crm");
$outDir = "/tmp/crm-export";

if (!is_dir($outDir) && !mkdir($outDir, 0775, true)) {
    fwrite(STDERR, "ERROR: Cannot create directory " . $outDir . "\n");
    exit(1);
}

$collections = array(
    "database",
    "client",
    "list",
    "clientHistory",
    "databasePerm",
    "relation_type"
);

foreach ($collections as $name) {
    echo "Exporting " . $name . "...\n";

    $collection = $db->selectCollection($name);
    $cursor = $collection->find();

    $file = $outDir . "/" . $name . ".jsonl";
    $tmpFile = $file . ".part";
    $handle = fopen($tmpFile, "wb");

    if (!$handle) {
        fwrite(STDERR, "ERROR: Cannot open " . $tmpFile . "\n");
        exit(1);
    }

    $count = 0;

    foreach ($cursor as $document) {
        $json = json_encode($document);

        if ($json === false) {
            fclose($handle);
            fwrite(STDERR, "ERROR: JSON encoding failed in " . $name .
                " near record " . ($count + 1) . "\n");
            exit(1);
        }

        if (fwrite($handle, $json . "\n") === false) {
            fclose($handle);
            fwrite(STDERR, "ERROR: Write failed: " . $tmpFile . "\n");
            exit(1);
        }

        $count++;

        if ($count % 100000 === 0) {
            echo "  " . $name . ": " . $count . " documents\n";
        }
    }

    fclose($handle);

    if (!rename($tmpFile, $file)) {
        fwrite(STDERR, "ERROR: Cannot finalize " . $file . "\n");
        exit(1);
    }

    echo "DONE: " . $name . " — " . $count . " documents; " .
        $file . "\n";
}

echo "EXPORT COMPLETE: " . $outDir . "\n";

function legacyDateTime(array $row, string $field): string
{
    $value = $row[$field] ?? null;
    if (!$value) {

}

function legacyTimestamp(array $row, string $field): int

    return ($value === null || $value === '') ? 0 : max(0, (int)$value);

                $skippedValues++;
                continue;

                $skippedValues++;
                continue;

    exit(1);

/*
 * Пакетная вставка.
 * При ошибке SQL выполняется fallback по одной записи,
 * чтобы изолировать проблемные строки и продолжить импорт.
 */
function flushBatch(
    PDO $pdo,
    string $table,
    array $columns,
    array &$batch,
    int &$imported,
    int &$skipped
): void {
    if (!$batch) {
        return;
    }

    $rows = $batch;
    $batch = [];

    try {
        $sql = makeInsertSql($table, $columns, count($rows));
        $stmt = $pdo->prepare($sql);

        $params = [];
        foreach ($rows as $item) {
            foreach ($item['data'] as $value) {
                $params[] = $value;
            }
        }

        $stmt->execute($params);
        $inserted = $stmt->rowCount();
        $imported += $inserted;
        $skipped += count($rows) - $inserted;
        return;
    } catch (Throwable $e) {
        logError(
            "Пачка {$table}, строк источника "
            . $rows[0]['line'] . '-' . $rows[count($rows) - 1]['line']
            . ': ' . $e->getMessage()
            . '; повторная проверка по одной записи'
        );
    }

    $placeholders = implode(',', array_fill(0, count($columns), '?'));
    $sql = 'INSERT IGNORE INTO `' . $table . '` (`'
        . implode('`,`', $columns) . '`) VALUES (' . $placeholders . ')';
    $stmt = $pdo->prepare($sql);

    foreach ($rows as $item) {
        try {
            $stmt->execute($item['data']);

            if ($stmt->rowCount() > 0) {
                $imported++;
            } else {
                $skipped++;
            }
        } catch (Throwable $e) {
            $skipped++;
            logError(
                "Таблица {$table}, строка источника {$item['line']}: "
                . $e->getMessage()
            );
        }
    }
}
